import ifj26.*
def gcd(a: Int, b: Int): Int =
    if a == b then
        return a
    else
        if a > b then
            return gcd(a - b, b)
        else
            return gcd(a, b - a)
def evenSum(n: Int): Int =
    if n <= 0 then
        return 0
    else
        return n + oddSum(n - 1)
def oddSum(n: Int): Int =
    if n <= 0 then
        return 0
    else
        return 2 * n + evenSum(n - 1)
def blend(
x: Int,
y: Double,
z: Int
): Double = -(x - z * 2) + y / 2
def forwardInferred(k: Int) = helperTail(k) + 0.5
def helperTail(m: Int) =
    val tmp = m * 3 - 1
    tmp
def stringCheck(prefix: String, code: Int): Int =
    val full = prefix + chr(code) + "26"
    val part = substr(full, 0, 4)
    val len = strLength(full)
    val firstOrd = ord(part, 0)
    if part < full then
        if part != full then
            return len * 100 + firstOrd
        else
            return -1
    else
        return -2
